```301:332:modules/rag/vector_store.py
    def _search_in(self, name: str, query_vec: list, k: int, expr: Optional[str]) -> list:
        """在单个集合里做一次 ANN 检索并整理结果。"""
        collection = self._get_collection(name)
        base_fields = ["chunk_id", "doc_id", "content", "source", "file_name", "updated_at"]
        extra_fields = ["kb_code", "dept_code"] if self._has_field(name, "kb_code") else []
        try:
            results = self._search_once(collection, query_vec, k, expr, base_fields + extra_fields)
        except Exception as e:
            if not expr:
                raise
            _logger.warning(f"集合 {name} 过滤检索失败，回退无过滤（疑似集合缺 kb_code 字段）: {e}")
            results = self._search_once(collection, query_vec, k, None, base_fields)

        kb = self.kb_of_collection(name)
        hits = []
        for hit in results[0]:   # type: ignore
            score = hit.score
            if score < self._threshold:
                _logger.debug(f"低相关结果过滤: score={score:.4f} < {self._threshold}")
                continue
            hits.append({
                "content": hit.entity.get("content", ""),
                ...
                "score": round(score, 4),
                "kb_code": hit.entity.get("kb_code", kb) if "kb_code" in extra_fields else kb,
                "collection": name,
            })
        return hits
```

```423:434:modules/rag/vector_store.py
    def _search_once(self, collection, query_vec: list, k: int, expr: Optional[str], output_fields: list):
        """执行一次 ANN 检索；expr 为空时不下过滤条件（兼容老集合）。"""
        kwargs = dict(
            data=[query_vec],
            anns_field="embedding",
            param={"metric_type": "COSINE", "params": {"nprobe": 16}},
            limit=k,
            output_fields=output_fields,
        )
        if expr:
            kwargs["expr"] = expr
        return collection.search(**kwargs)
```

## ⑩ 时效加权 + 精排

```103:114:modules/rag/rag_chain.py
    @staticmethod
    def _apply_recency_boost(candidates: list) -> None:
        """边界5：按文档更新时间加权，30天内 +0.05、90天内 +0.02，避免旧版文档压过新版。"""
        now = datetime.now()
        for c in candidates:
            try:
                days = (now - datetime.strptime(c.get("updated_at", ""), "%Y-%m-%d %H:%M:%S")).days
            except Exception:
                continue
            bonus = 0.05 if days <= 30 else (0.02 if days <= 90 else 0.0)
            if bonus:
                c["score"] = round(c.get("score", 0.0) + bonus, 4)
```

```102:114:modules/rag/reranker.py
    def _rerank_fallback(self, query: str, documents: List[dict], top_k: int) -> List[dict]:
        """降级精排：向量相似度 + 关键词命中加权。"""
        query_terms = set(query.lower().split())
        for doc in documents:
            content = doc.get("content", "").lower()
            term_hits = sum(1 for t in query_terms if t in content)
            term_score = term_hits / max(len(query_terms), 1)
            vec_score = doc.get("score", 0.5)
            doc["rerank_score"] = round(vec_score * 0.7 + term_score * 0.3, 4)
        documents.sort(key=lambda x: x["rerank_score"], reverse=True)
        result = documents[:top_k]
```

```116:135:modules/rag/rag_chain.py
    def build_context(
        self,
        query: str,
        top_k: Optional[int] = None,
        expr: Optional[str] = None,
        kb_codes: Optional[list] = None,
    ) -> str:
        docs = self.retrieve(query, top_k=top_k, expr=expr, kb_codes=kb_codes)
        if not docs:
            return ""
        blocks = []
        for i, doc in enumerate(docs, 1):
            score = doc.get("rerank_score", doc.get("score", 0))
```
